# Skill Retrieval + Per-Knowledge-Point Lesson Pipeline

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Refactor lesson generation into a two-step pipeline (retrieve skills, then generate content) that displays pedagogy skill + knowledge point skill separately and generates lesson content per knowledge point.

**Architecture:** Two skill layers with a strict hierarchy:

1. **教学法 skill (pedagogy)** — the parent layer. **Mandatory**: user must select a pedagogy skill before the learning flow (knowledge graph → quiz → lesson) can proceed. Retrieved from `skills/{name}/SKILL.md`.
2. **知识点 skill (knowledge point)** — the child layer. **Optional**: user can toggle on/off via `use_knowledge_skills`. When enabled, LLM generates a teaching micro-skill for each weak knowledge point, derived from the active pedagogy method. When disabled, lesson generation uses only the pedagogy skill (current behavior).

Pipeline: Step 1 retrieves pedagogy skill + optionally generates knowledge point skills. Step 2 iterates weak knowledge points to generate lesson sections. Both skill types are visible in the UI as independent badges.

**Tech Stack:** Python/FastAPI backend, Jinja2 prompts, Instructor for structured output, vanilla JS + Tailwind frontend.

---

## Hierarchy & Constraints

```
教学法 skill (mandatory, selected by user)
  ├── feynman / socratic / scaffolding
  └── Determines HOW to teach
        │
        ▼  (derives)
知识点 skill (optional, generated per weak point)
  ├── Generated by LLM using pedagogy method + knowledge point info
  └── Determines HOW to teach THIS SPECIFIC knowledge point
        │
        ▼  (informs)
Lesson content (per knowledge point)
```

**Enforcement rules:**
- The learning flow (`learn_topic` intent → knowledge graph → quiz → lesson) **requires** an active pedagogy skill. If none is selected, the system prompts the user to choose one before proceeding.
- Knowledge point skills are **opt-in**: controlled by `use_knowledge_skills` flag in state (default: `true`). When `false`, Step 1b is skipped and lesson generation uses only the pedagogy skill.
- The `knowledge_skill.j2` prompt always receives the pedagogy method + instructions — ensuring the generated knowledge point skill is consistent with the parent pedagogy approach.

```python
from __future__ import annotations

import logging
from src.agent.state import LearningState
from src.llm.client import get_raw_client, get_model_name
from src.prompts.loader import render_prompt
from src.skills.loader import get_skill_section
from langchain_core.messages import AIMessage

logger = logging.getLogger(__name__)


def generate_knowledge_skill(
    topic: str,
    knowledge_point_name: str,
    knowledge_point_description: str,
    pedagogy_method: str,
    pedagogy_instructions: str,
) -> str:
    """Step 1b: Generate a teaching micro-skill for one knowledge point via LLM."""
    prompt = render_prompt(
        "knowledge_skill",
        topic=topic,
        knowledge_point_name=knowledge_point_name,
        knowledge_point_description=knowledge_point_description,
        pedagogy_method=pedagogy_method,
        pedagogy_instructions=pedagogy_instructions,
    )
    client = get_raw_client()
    response = client.chat.completions.create(
        model=get_model_name(),
        max_tokens=512,
        messages=[{"role": "user", "content": prompt}],
    )
    return response.choices[0].message.content.strip()


def retrieve_skills(state: dict) -> tuple[str, dict[str, str], list[dict]]:
    """Step 1: Retrieve pedagogy skill + generate knowledge point skills.

    Returns:
        (pedagogy_instructions, knowledge_skills, weak_points)
        - pedagogy_instructions: str from active skill's lesson section
        - knowledge_skills: {knowledge_point_name: generated_skill_text}
        - weak_points: list of {"name": ..., "description": ...} for weak points
    """
    diagnosis = state.get("diagnosis_result")
    graph = state.get("knowledge_graph")
    topic = state.get("topic", "")

    # Step 1a: Retrieve pedagogy skill
    pedagogy_instructions = get_skill_section(state, "lesson")

    if not diagnosis or not graph:
        return pedagogy_instructions, {}, []

    # Extract weak points
    breakdown = diagnosis.breakdown if hasattr(diagnosis, "breakdown") else diagnosis.get("breakdown", [])
    nodes = graph.nodes if hasattr(graph, "nodes") else graph.get("nodes", [])
    node_map = {(n.name if hasattr(n, "name") else n["name"]): n for n in nodes}

    weak_items = [b for b in breakdown if (b.status if hasattr(b, "status") else b["status"]) == "weak"]

    weak_points = []
    for item in weak_items:
        name = item.name if hasattr(item, "name") else item["name"]
        node = node_map.get(name)
        desc = ""
        if node:
            desc = node.description if hasattr(node, "description") else node.get("description", "")
        weak_points.append({"name": name, "description": desc})

    if not weak_points:
        return pedagogy_instructions, {}, []

    # Step 1b: Generate knowledge point skills (optional — user can toggle off)
    if not state.get("use_knowledge_skills", True):
        return pedagogy_instructions, {}, weak_points

    active_skill = state.get("active_skill", "")
    skill_meta = None
    if active_skill:
        from src.skills.loader import get_skill
        skill_meta = get_skill(active_skill)

    pedagogy_method = skill_meta.display_name if skill_meta else "默认教学法"

    knowledge_skills = {}
    for wp in weak_points:
        skill_text = generate_knowledge_skill(
            topic=topic,
            knowledge_point_name=wp["name"],
            knowledge_point_description=wp["description"],
            pedagogy_method=pedagogy_method,
            pedagogy_instructions=pedagogy_instructions or "无特定教学法要求",
        )
        knowledge_skills[wp["name"]] = skill_text

    return pedagogy_instructions, knowledge_skills, weak_points


def generate_lesson_per_point(
    topic: str,
    weak_points: list[dict],
    pedagogy_instructions: str,
    knowledge_skills: dict[str, str],
) -> str:
    """Step 2: Generate lesson content per knowledge point, then concatenate.

    Each knowledge point gets its own LLM call with both the pedagogy
    skill and its generated knowledge point skill as context.
    Returns the concatenated lesson markdown.
    """
    client = get_raw_client()
    sections = []
    total = len(weak_points)

    for i, wp in enumerate(weak_points):
        prompt = render_prompt(
            "lesson_point",
            topic=topic,
            knowledge_point_name=wp["name"],
            knowledge_point_description=wp["description"],
            pedagogy_instructions=pedagogy_instructions or "无特定教学法要求",
            knowledge_skill=knowledge_skills.get(wp["name"], "无额外教学技能"),
            point_index=i + 1,
            total_points=total,
        )
        response = client.chat.completions.create(
            model=get_model_name(),
            max_tokens=4096,
            messages=[{"role": "user", "content": prompt}],
        )
        sections.append(response.choices[0].message.content.strip())

    return "\n\n---\n\n".join(sections)

def lesson_node(state: LearningState) -> dict:
    """Generate personalized lesson using two-step pipeline."""
    diagnosis = state.get("diagnosis_result")
    graph = state.get("knowledge_graph")
    topic = state.get("topic", "")

    if not diagnosis:
        return {"mode": "chat", "messages": [AIMessage(content="请先完成诊断。")]}

    nodes = graph.nodes if hasattr(graph, "nodes") else graph.get("nodes", [])

    # Step 1: Retrieve skills
    pedagogy_instructions, knowledge_skills, weak_points = retrieve_skills(state)

    if not weak_points:
        # All mastered — review lesson (no knowledge skills needed)
        lesson_text = _generate_review_lesson(topic, nodes, pedagogy_instructions)
        return {
            "lesson_content": lesson_text,
            "knowledge_skills": None,
            "mode": "lesson",
            "messages": [AIMessage(content=lesson_text)],
        }

    # Step 2: Generate per-point content
    lesson_text = generate_lesson_per_point(topic, weak_points, pedagogy_instructions, knowledge_skills)

    return {
        "lesson_content": lesson_text,
        "knowledge_skills": knowledge_skills,
        "mode": "lesson",
        "messages": [AIMessage(content=lesson_text)],
    }


def _build_lesson_prompt(state: dict) -> str | None:
    """Build the lesson prompt from state.

    DEPRECATED: Only used for backward compatibility with old callers.
    New code should use retrieve_skills() + generate_lesson_per_point().
    Returns None if diagnosis missing.
    """
    diagnosis = state.get("diagnosis_result")
    graph = state.get("knowledge_graph")
    topic = state.get("topic", "")

    if not diagnosis:
        return None

    breakdown = diagnosis.breakdown if hasattr(diagnosis, "breakdown") else diagnosis.get("breakdown", [])
    weak_items = [b for b in breakdown if (b.status if hasattr(b, "status") else b["status"]) == "weak"]

    nodes = graph.nodes if hasattr(graph, "nodes") else graph.get("nodes", [])
    node_map = {(n.name if hasattr(n, "name") else n["name"]): n for n in nodes}

    skill_instructions = get_skill_section(state, "lesson")

    if not weak_items:
        all_points = [{"name": n.name if hasattr(n, "name") else n["name"],
                       "description": n.description if hasattr(n, "description") else n.get("description", "")}
                      for n in nodes]
        return render_prompt("lesson_review", topic=topic, points=all_points, skill_instructions=skill_instructions)

    weak_points = []
    for item in weak_items:
        name = item.name if hasattr(item, "name") else item["name"]
        node = node_map.get(name)
        desc = node.description if node and hasattr(node, "description") else (node.get("description", "") if node else "")
        weak_points.append({"name": name, "description": desc})
    return render_prompt("lesson_plan", topic=topic, weak_points=weak_points, skill_instructions=skill_instructions)


def lesson_node_stream(state: dict):

            model=get_model_name(),

  } else if (progress && progress.includes('/')) {
    const [done, total] = progress.split('/');
    el.textContent = `正在生成内容（${done}/${total} 个知识点）...`;
  } else {
    el.textContent = '正在生成教案...';
  }
}
```
